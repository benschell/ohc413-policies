% --- Fillable form fields -------------------------------------------------
% Fields are real AcroForm widgets (typeable in any PDF reader) drawn with an
% underline border, so the same PDF also prints as a normal fill-in-by-hand
% form. Keep hyperref last-ish: it patches a lot of internals.
\usepackage[hidelinks]{hyperref}

% hyperref wants an RGB triple, not a named color.
\providecommand{\formbordercolor}{0.09 0.243 0.51}

\newlength{\ffboxw}
\newlength{\fflabelw}
\newlength{\fffieldw}

% \formfield{name}{label}{width fraction}{height}
\newcommand{\formfield}[4]{%
  \setlength{\ffboxw}{#3\textwidth}%
  \settowidth{\fflabelw}{\textbf{#2:}}%
  \setlength{\fffieldw}{\ffboxw - \fflabelw - 0.5em}%
  \makebox[\ffboxw][l]{%
    \textbf{#2:}\hspace{0.5em}%
    \TextField[
      name=#1,
      width=\fffieldw,
      height=#4,
      borderstyle=U,
      borderwidth=1,
      bordercolor={\formbordercolor},
      backgroundcolor={},
      color={\formbordercolor},
      charsize=10pt
    ]{}%
  }%
}

% \formfieldstacked{name}{label}{width fraction}{height}
% Label on its own line above a full-width box — for long labels where an
% inline field would leave no room to write.
\newcommand{\formfieldstacked}[4]{%
  \begin{minipage}[t]{#3\textwidth}%
    \textbf{#2:}\par\vspace{0.2em}%
    \TextField[
      name=#1,
      width=#3\textwidth,
      height=#4,
      borderstyle=U,
      borderwidth=1,
      bordercolor={\formbordercolor},
      backgroundcolor={},
      color={\formbordercolor},
      charsize=10pt
    ]{}%
  \end{minipage}%
}

% \formcheckbox{name}{label}{width fraction}
\newcommand{\formcheckbox}[3]{%
  \begin{minipage}[t]{#3\textwidth}%
    \setlength{\parindent}{0pt}%
    \hangindent=1.6em\hangafter=1%
    \CheckBox[
      name=#1,
      width=10pt,
      height=10pt,
      bordercolor={\formbordercolor},
      backgroundcolor={},
      checkboxsymbol=\ding{51}
    ]{}\hspace{0.6em}#2%
  \end{minipage}%
}
