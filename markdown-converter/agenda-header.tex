}

  \renewcommand{\headrulewidth}{4pt}
  \renewcommand{\headrule}{\color{\agendabarcolor}\hrule height 4pt}
  \renewcommand{\footrulewidth}{4pt}
  \renewcommand{\footrule}{\color{\agendabarcolor}\hrule height 4pt}
  \fancyfoot[C]{\small\color{ScoutingWarmGray}\thepage}
}
