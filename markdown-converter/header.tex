% Scouting America Brand Colors
\definecolor{ScoutingRed}{HTML}{CE1126}
\definecolor{ScoutingDarkBlue}{HTML}{003366}
\definecolor{ScoutingWarmGray}{HTML}{515354}

% Fonts - Roboto family
\usepackage{fontspec}
\setmainfont{Roboto Condensed}[
  UprightFont=*-Regular,
  BoldFont=*-Bold,
  ItalicFont=*-Italic
]

\newfontfamily\robotoslab{Roboto Slab}[
  UprightFont=*-Regular,
  BoldFont=*-Bold
]

% Title formatting
\usepackage{titlesec}
\titleformat{\section}
  {\robotoslab\Large\bfseries\color{ScoutingWarmGray}}
  {}
  {0em}
  {}

\titleformat{\subsection}
  {\robotoslab\large\bfseries\color{ScoutingDarkBlue}}
  {}
  {0em}
  {}

\titleformat{\subsubsection}
  {\bfseries\normalsize\color{ScoutingDarkBlue}}
  {}
  {0em}
  {}

% Table styling
\usepackage{colortbl}
\arrayrulecolor{ScoutingDarkBlue}

% Footer only - no header
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\fancyfoot[C]{\small\color{ScoutingWarmGray}\thepage}

% Plain style for first page
\fancypagestyle{plain}{
  \fancyhf{}
  \renewcommand{\headrulewidth}{0pt}
  \fancyfoot[C]{\small\color{ScoutingWarmGray}\thepage}
}

% Enhanced title formatting (27pt, centered)
\usepackage{titling}
\pretitle{\begin{center}\robotoslab\fontsize{27}{32}\bfseries\color{ScoutingWarmGray}}

\postauthor{\end{center}}
